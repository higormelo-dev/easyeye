<script setup>
import { ref, watch } from 'vue';
import OffcanvasPanel from '@/Components/Panel/OffcanvasPanel.vue';
 
/**
 * Drawer de detalhes do Equipamento (read-only).
 */
const props = defineProps({
    open:    { type: Boolean, required: true },
    showUrl: { type: String,  default: '' },
    t:       { type: Object,  default: () => ({}) },
});
 
defineEmits(['close']);
 
const loading = ref(false);
const item    = ref(null);
 
async function loadDetail(url) {
    loading.value = true;
    item.value    = null;
    try {
        const res  = await fetch(url, { headers: { Accept: 'application/json' } });
        const json = await res.json();
        item.value = json.data;
    } finally {
        loading.value = false;
    }
}
 
watch(() => props.open, (val) => {
    if (val && props.showUrl) loadDetail(props.showUrl);
    if (!val) item.value = null;
});
</script>
 
<template>
    <OffcanvasPanel
        :open="open"
        :width="480"
        :loading="loading"
        :loading-label="t.detail_loading"
        @close="$emit('close')"
    >
        <!-- Header -->
        <template #header>
            <div>
                <h5 class="mb-0 fw-semibold">
                    <i class="ti ti-device-laptop me-2 text-info"></i>
                    {{ item?.name ?? t.detail_loading }}
                </h5>
                <code v-if="item" class="text-muted small">{{ item.code }}</code>
            </div>
        </template>
 
        <!-- Body -->
        <template v-if="item">
            <!-- Status -->
            <div class="mb-4">
                <span v-if="item.deleted" class="badge bg-secondary">
                    {{ t.status_deleted ?? 'Removido' }}
                </span>
                <span v-else-if="item.active" class="badge bg-success">
                    {{ t.status_active ?? 'Ativo' }}
                </span>
                <span v-else class="badge bg-danger">
                    {{ t.status_inactive ?? 'Inativo' }}
                </span>
            </div>
 
            <!-- Section: Identidade -->
            <div class="detail-section">
                <div class="detail-section__title">
                    <i class="ti ti-id-badge-2 me-1"></i> {{ t.detail_section_identity ?? 'Identidade' }}
                </div>
                <div class="detail-table">
                    <div class="detail-row">
                        <span class="detail-label">{{ t.col_code ?? 'Código' }}</span>
                        <span class="detail-value"><code>{{ item.code }}</code></span>
                    </div>
                    <div class="detail-row">
                        <span class="detail-label">{{ t.col_name ?? 'Nome' }}</span>
                        <span class="detail-value">{{ item.name }}</span>
                    </div>
                    <div class="detail-row">
                        <span class="detail-label">{{ t.col_serial ?? 'Nº Série' }}</span>
                        <span class="detail-value"><code>{{ item.serial_number || '—' }}</code></span>
                    </div>
                </div>
            </div>
 
            <!-- Section: Rede -->
            <div class="detail-section">
                <div class="detail-section__title">
                    <i class="ti ti-network me-1"></i> {{ t.detail_section_network ?? 'Rede' }}
                </div>
                <div class="detail-table">
                    <div class="detail-row">
                        <span class="detail-label">{{ t.col_ip ?? 'IP' }}</span>
                        <span class="detail-value"><code>{{ item.ip || '—' }}</code></span>
                    </div>
                    <div class="detail-row">
                        <span class="detail-label">{{ t.col_mac ?? 'MAC' }}</span>
                        <span class="detail-value"><code>{{ item.mac || '—' }}</code></span>
                    </div>
                </div>
            </div>
 
            <!-- Section: Auditoria -->
            <div class="detail-section">
                <div class="detail-section__title">
                    <i class="ti ti-history me-1"></i> {{ t.detail_section_audit ?? 'Auditoria' }}
                </div>
                <div class="detail-table">
                    <div class="detail-row">
                        <span class="detail-label">{{ t.detail_registered_at ?? 'Cadastrado em' }}</span>
                        <span class="detail-value">{{ item.created_at || '—' }}</span>
                    </div>
                    <div v-if="item.deleted_at" class="detail-row">
                        <span class="detail-label">{{ t.detail_deleted_at ?? 'Removido em' }}</span>
                        <span class="detail-value">{{ item.deleted_at }}</span>
                    </div>
                </div>
            </div>
        </template>
    </OffcanvasPanel>
</template>
